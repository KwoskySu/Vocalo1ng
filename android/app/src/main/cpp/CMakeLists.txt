cmake_minimum_required(VERSION 3.22.1)
project(vocalo_runtime)
if(NOT DEFINED NODE_MOBILE_ROOT)
  message(FATAL_ERROR "Run tools/prepare-android-runtime.ps1 first")
endif()
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${NODE_MOBILE_ROOT}/bin/${ANDROID_ABI}/libnode.so")
add_library(vocalo_runtime SHARED runtime.cpp)
target_link_libraries(vocalo_runtime node log)
target_link_options(vocalo_runtime PRIVATE "-Wl,-z,max-page-size=16384")
